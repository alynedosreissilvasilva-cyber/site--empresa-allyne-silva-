-- 1) TABELA DE FUNCIONARIOS
create table if not exists public.funcionarios (
    id           bigint generated by default as identity primary key,
    nome         text not null,
    email        text not null,
    cargo        text,
    departamento text,
    foto_url     text,
    created_at   timestamptz default now()
);

-- 2) HABILITA ROW LEVEL SECURITY
alter table public.funcionarios enable row level security;

-- 3) POLICAS DE ACESSO PARA O SITE (papel anon)
drop policy if exists "anon_select_funcionarios" on public.funcionarios;
create policy "anon_select_funcionarios"
    on public.funcionarios for select
    to anon, authenticated
    using (true);

drop policy if exists "anon_insert_funcionarios" on public.funcionarios;
create policy "anon_insert_funcionarios"
    on public.funcionarios for insert
    to anon, authenticated
    with check (true);

drop policy if exists "anon_update_funcionarios" on public.funcionarios;
create policy "anon_update_funcionarios"
    on public.funcionarios for update
    to anon, authenticated
    using (true)
    with check (true);

drop policy if exists "anon_delete_funcionarios" on public.funcionarios;
create policy "anon_delete_funcionarios"
    on public.funcionarios for delete
    to anon, authenticated
    using (true);

-- 4) BUCKET PUBLICO DAS FOTOS
insert into storage.buckets (id, name, public)
values ('fotos-funcionarios', 'fotos-funcionarios', true)
on conflict (id) do update set public = true;

-- 5) POLICAS DE UPLOAD/LEITURA NO BUCKET
drop policy if exists "upload_fotos_funcionarios" on storage.objects;
create policy "upload_fotos_funcionarios"
    on storage.objects for insert
    to anon, authenticated
    with check (bucket_id = 'fotos-funcionarios');

drop policy if exists "select_fotos_funcionarios" on storage.objects;
create policy "select_fotos_funcionarios"
    on storage.objects for select
    to anon, authenticated
    using (bucket_id = 'fotos-funcionarios');
